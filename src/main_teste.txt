#include <Wire.h>
#include "hardware/Motor/Motor.h"
#include "hardware/SonarSensor/SonarSensor.h"
#include <Arduino.h>
#include <WiFi.h>

void setup()
{
  Serial.begin(115200);
  WiFi.disconnect(true); // desconecta e limpa config
  WiFi.mode(WIFI_OFF);   // desliga rádio Wi-Fi
  btStop();

  pinMode(5, OUTPUT);
  pinMode(6, OUTPUT);
  pinMode(7, OUTPUT);

  pinMode(9, OUTPUT);
  pinMode(10, INPUT);
  pinMode(11, INPUT);

  ledcSetup(0, 490, 8);
  ledcAttachPin(5, 0);
}

  int pwm = map(pot, 0, 4095, 0, 255);
